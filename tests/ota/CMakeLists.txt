cmake_minimum_required(VERSION 3.16)
project(note4c_ota_tests LANGUAGES C CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
set(ENABLE_CJSON_TEST OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(cjson GIT_REPOSITORY https://github.com/DaveGamble/cJSON.git
    GIT_TAG c859b25da02955fef659d658b8f324b5cde87be3)
FetchContent_MakeAvailable(cjson)
set(OTA ../../firmware/components/lucidcairn_ota)
add_executable(ota_policy_test ota_policy_test.cc ${OTA}/ota_policy.cc)
target_include_directories(ota_policy_test PRIVATE ${OTA}/include ${cjson_SOURCE_DIR})
target_link_libraries(ota_policy_test PRIVATE cjson)
add_executable(ota_health_test ota_health_test.cc ${OTA}/ota_health.cc)
target_include_directories(ota_health_test PRIVATE fakes ${OTA}/include)
foreach(target ota_policy_test ota_health_test)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
        option(OTA_SANITIZERS "Run host checks with memory/undefined behavior sanitizers" ON)
        if(OTA_SANITIZERS)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
    add_test(NAME ${target} COMMAND ${target})
endforeach()
